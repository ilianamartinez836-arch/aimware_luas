local ref = gui.Reference("Visuals", "World")
local box = gui.Groupbox(ref, "Distance Lines PRO", 320, 350, 305, 0)

local enable = gui.Checkbox(box, "dist.lines.enable5", "Enable Distance Lines", false)
local from_crosshair = gui.Checkbox(box, "dist.lines.crosshair5", "Draw From Crosshair", true)
local from_bottom_screen = gui.Checkbox(box, "dist.lines.screenbottom5", "Draw From Screen Bottom", false)
local show_distance = gui.Checkbox(box, "dist.lines.text5", "Show Distance Text", true)
local colorpicker = gui.ColorPicker(box, "dist.lines.color5", "Line Color", 255, 0, 0, 200)

local font = draw.CreateFont("Tahoma", 14)

local function SafeGetProp(ent, prop)
    local ok, val = pcall(function() return ent:GetProp(prop) end)
    if not ok then return nil end
    return val
end

local function IsGhost(ent)
    local g = SafeGetProp(ent, "m_isGhost")
    return (g == 1 or g == true)
end

local function IsDormantSafe(ent)
    local ok, val = pcall(function() return ent:IsDormant() end)
    if not ok then return false end
    return val == true
end

local function GetPlayers()
    local p = entities.FindByClass("CTerrorPlayer")
    if not p or #p == 0 then
        p = entities.FindByClass("CCSPlayer")
    end
    return p
end

callbacks.Register("Draw", function()
    if not enable:GetValue() then return end

    local me = entities.GetLocalPlayer()
    if not me or not me:IsAlive() then return end

    local myTeam = me:GetTeamNumber()
    local myPos = me:GetAbsOrigin()

    local screen_w, screen_h = draw.GetScreenSize()
    local start_x, start_y = screen_w / 2, screen_h / 2

    if from_bottom_screen:GetValue() then
        start_x = screen_w / 2
        start_y = screen_h - 1
    elseif from_crosshair:GetValue() then
        start_x = screen_w / 2
        start_y = screen_h / 2
    end

    local r, g, b, a = colorpicker:GetValue()

    local players = GetPlayers()
    if not players then return end

    draw.SetFont(font)

    local closestDist = 999999

    for _, player in pairs(players) do
        if player ~= me and player:IsAlive() and not IsDormantSafe(player) then

            local tTeam = player:GetTeamNumber()
            if myTeam == 2 then
                if tTeam ~= 3 then goto continue end
                if IsGhost(player) then goto continue end
            end
            if myTeam == 3 then
                if tTeam ~= 2 then goto continue end
            end

            local pos = player:GetAbsOrigin() + Vector3(0,0,60)
            local dist_units = (pos - myPos):Length()
            local dist_meters = math.floor(dist_units * 0.01905)

            if dist_meters < closestDist then
                closestDist = dist_meters
            end

            local sx, sy = client.WorldToScreen(pos)
            if sx and sy then
                draw.Color(r, g, b, a)
                draw.Line(start_x, start_y, sx, sy)

                if show_distance:GetValue() then
                    draw.Color(255,255,255,255)
                    draw.Text(sx - 10, sy - 15, dist_meters .. "m")
                end
            end
        end
        ::continue::
    end

    if closestDist ~= 999999 then
        draw.Color(0,0,0,150)
        draw.FilledRect(screen_w - 140, 20, screen_w - 20, 60)
        draw.Color(255,255,255,255)
        draw.Text(screen_w - 130, 30, "Closest Target:")
        draw.Text(screen_w - 130, 45, closestDist .. "m")
    end

end)

local ref = gui.Reference("Visuals", "Other")
local gb  = gui.Groupbox(ref, "Melee Cooldown", 325, 315, 300, 0)

local enable   = gui.Checkbox(gb, "mcd.enable", "Enable Melee Cooldown", true)
local bar_col  = gui.ColorPicker(gb, "mcd.col", "Bar Rengi", 255, 200, 0, 255)

local font = draw.CreateFont("Tahoma", 12, 600)

local MELEE_COOLDOWN = 0.5 -- saniye (yaklaşık)
local last_swing_time = 0
local was_swinging    = false

callbacks.Register("Draw", "melee_cd", function()
    if not enable:GetValue() then return end

    local lp = entities.GetLocalPlayer()
    if not lp or not lp:IsAlive() then return end

    local wep = lp:GetPropEntity("m_hActiveWeapon")
    if not wep then return end

    local cls = wep:GetClass()
    if cls ~= "CTerrorMeleeWeapon" and cls ~= "CChainsaw" then return end

    local swinging = wep:GetPropInt("m_bInMeleeSwing") == 1
    local now      = globals.RealTime()

    if swinging and not was_swinging then
        last_swing_time = now
    end
    was_swinging = swinging

    local elapsed = now - last_swing_time
    local ratio   = math.min(elapsed / MELEE_COOLDOWN, 1.0)

    local sw, sh = draw.GetScreenSize()
    local barW   = 200
    local barH   = 8
    local barX   = sw/2 - barW/2
    local barY   = sh/2 + 40

    draw.Color(0, 0, 0, 180)
    draw.FilledRect(barX, barY, barX + barW, barY + barH)

    local r, g, b, _ = bar_col:GetValue()
    if ratio < 1.0 then
        -- Dolmadı — kırmızıdan renge geçiş
        local tr = math.floor(255 * (1 - ratio))
        local tg = math.floor(255 * ratio)
        draw.Color(tr, tg, 0, 220)
    else
        draw.Color(r, g, b, 220)
    end
    draw.FilledRect(barX, barY, barX + math.floor(barW * ratio), barY + barH)

    draw.Color(0, 0, 0, 200)
    draw.OutlinedRect(barX, barY, barX + barW, barY + barH)

    draw.SetFont(font)
    local label = ratio >= 1.0 and "READY" or string.format("%.1fs", MELEE_COOLDOWN - elapsed)
    local tw, _ = draw.GetTextSize(label)
    draw.Color(0, 0, 0, 180)
    draw.Text(sw/2 - tw/2 + 1, barY - 15 + 1, label)
    draw.Color(ratio >= 1.0 and 100 or 255, ratio >= 1.0 and 255 or 200, 100, 255)
    draw.Text(sw/2 - tw/2, barY - 15, label)
end)

===== END: Melee Cooldown Bar =====

ffi.cdef[[
    void* GetModuleHandleA(const char* lpModuleName);
    int   VirtualProtect(void* lpAddress, size_t dwSize, uint32_t flNewProtect, uint32_t* lpflOldProtect);
]]

local ref       = gui.Reference("Misc", "General", "Extra")
local gl_enable = gui.Checkbox(ref, "gl.enable", "Glow Rengi Aktif",    false)
local gl_color  = gui.ColorPicker(ref, "gl.color", "Glow Rengi",         0, 255, 0, 255)
local gl_entity = gui.Checkbox(ref, "gl.entity", "Entity Bazlı (VMT)",  false)

local client_module = ffi.cast("uintptr_t", ffi.C.GetModuleHandleA("client.dll"))
if client_module == 0 then print("[Glow] client.dll none") return end
print(string.format("[Glow] client.dll: 0x%X", tonumber(client_module)))

local pMatSys = ffi.cast("void***", client_module + 0x88B7F0)[0]
if pMatSys == nil or ffi.cast("uintptr_t", pMatSys) == 0 then
    print("[Glow] pMatSys alinamadi!")
    return
end
print(string.format("[Glow] pMatSys: 0x%X", tonumber(ffi.cast("uintptr_t", pMatSys))))


local FindMat_t = ffi.typeof(
    "void*(__fastcall*)(void*, void*, const char*, const char*, int, void*)"
)
local ColorMod_t = ffi.typeof(
    "void(__fastcall*)(void*, void*, float, float, float)"
)
local GlowFn_t = ffi.typeof(
    "void(__fastcall*)(void*, void*, float*, float*, float*, float*)"
)

local pGlowColor = nil

local function GetGlowMaterial()
    if pGlowColor ~= nil then return pGlowColor end
    local vt = ffi.cast("void***", pMatSys)[0]
    local ok, res = pcall(function()
        return ffi.cast(FindMat_t, vt[71])(
            pMatSys, nil, "dev/glow_color", "Other textures", 1, nil
        )
    end)
    if ok and res ~= nil and ffi.cast("uintptr_t", res) > 0x10000 then
        pGlowColor = res
        print(string.format("[Glow] dev/glow_color: 0x%X", tonumber(ffi.cast("uintptr_t", pGlowColor))))
    end
    return pGlowColor
end

local function ColorModulate(pMat, r, g, b)
    local vt = ffi.cast("void***", pMat)[0]
    ffi.cast(ColorMod_t, vt[28])(pMat, nil, r, g, b)
end


local GLOW_ENTRY_RVA = 0x5B6850
local glow_entry     = ffi.cast("uintptr_t*", client_module + GLOW_ENTRY_RVA)
local orig_glow_addr = glow_entry[0]

local tmp_r = ffi.new("float[1]")
local tmp_g = ffi.new("float[1]")
local tmp_b = ffi.new("float[1]")
local tmp_a = ffi.new("float[1]")

local GC = { hook_fn = nil }

local function Hooked_GetGlowColor(pThis, edx, r, g, b, a)
    ffi.cast(GlowFn_t, orig_glow_addr)(pThis, nil, tmp_r, tmp_g, tmp_b, tmp_a)

    if gl_entity:GetValue() and gl_enable:GetValue() then
        if tmp_a[0] > 0.0 then
            local cr, cg, cb = gl_color:GetValue()
            r[0] = cr / 255.0
            g[0] = cg / 255.0
            b[0] = cb / 255.0
            a[0] = tmp_a[0]  
        else
            r[0] = 0.0
            g[0] = 0.0
            b[0] = 0.0
            a[0] = 0.0
        end
    else
        r[0] = tmp_r[0]
        g[0] = tmp_g[0]
        b[0] = tmp_b[0]
        a[0] = tmp_a[0]
    end
end

GC.hook_fn = ffi.cast(GlowFn_t, Hooked_GetGlowColor)

local old = ffi.new("uint32_t[1]")
ffi.C.VirtualProtect(ffi.cast("void*", client_module + GLOW_ENTRY_RVA), 4, 0x40, old)
glow_entry[0] = ffi.cast("uintptr_t", GC.hook_fn)
ffi.C.VirtualProtect(ffi.cast("void*", client_module + GLOW_ENTRY_RVA), 4, old[0], old)

print("[Glow] VMT hook kuruldu (slot 239).")

callbacks.Register("Draw", "Glow_Draw", function()
    if not gl_enable:GetValue() then return end
    if gl_entity:GetValue() then return end  

    local pMat = GetGlowMaterial()
    if pMat == nil then return end

    local r, g, b = gl_color:GetValue()
    ColorModulate(pMat, r / 255.0, g / 255.0, b / 255.0)
end)

callbacks.Register("Unload", function()
    local old2 = ffi.new("uint32_t[1]")
    ffi.C.VirtualProtect(ffi.cast("void*", client_module + GLOW_ENTRY_RVA), 4, 0x40, old2)
    glow_entry[0] = orig_glow_addr
    ffi.C.VirtualProtect(ffi.cast("void*", client_module + GLOW_ENTRY_RVA), 4, old2[0], old2)

    if pGlowColor ~= nil then
        ColorModulate(pGlowColor, 1.0, 1.0, 1.0)
    end

    callbacks.Unregister("Draw", "Glow_Draw")
    print("[Glow] Restore edildi.")
end)
