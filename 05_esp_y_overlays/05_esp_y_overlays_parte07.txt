local ref = gui.Reference("Visuals", "Local")
local gbox = gui.Groupbox(ref, "NAZI", 0, 450, 300, 0)

local hide_game_cb = gui.Checkbox(gbox, "lua_swasticross_hide", "auswählen", false)
local outline_cb = gui.Checkbox(gbox, "lua_swasticross_outline", "Siyah Dış Çizgi Ekle", true)
local color = gui.ColorPicker(gbox, "lua_swasticross_col", "Renk", 255, 0, 0, 255)
local size = gui.Slider(gbox, "lua_swasticross_size", "Boyut", 10, 5, 50)
local thickness = gui.Slider(gbox, "lua_swasticross_thick", "Kalınlık", 1, 1, 5)

local last_state = nil

local function DrawCrosshairShape(cx, cy, s, t)
    local t_half = t / 2

    draw.FilledRect(cx - t_half, cy - s, cx + t_half, cy + s)
    draw.FilledRect(cx - s, cy - t_half, cx + s, cy + t_half)
    -- Kollar
    draw.FilledRect(cx, cy - s - t_half, cx + s, cy - s + t_half)
    draw.FilledRect(cx + s - t_half, cy, cx + s + t_half, cy + s)
    draw.FilledRect(cx - s, cy + s - t_half, cx, cy + s + t_half)
    draw.FilledRect(cx - s - t_half, cy - s, cx - s + t_half, cy)
end

callbacks.Register("Draw", "swastika_crosshair_logic", function()
    local active = hide_game_cb:GetValue()

    if active ~= last_state then
        if active then 

            client.Command("crosshair 0", false) 
        else 

            client.Command("crosshair 1", false) 
        end
        last_state = active
    end

    if active then
        local me = entities.GetLocalPlayer()
        if not me or not me:IsAlive() then return end

        local sw, sh = draw.GetScreenSize()
        local cx, cy = sw / 2, sh / 2
        
        local r, g, b, a = color:GetValue()
        local s = size:GetValue()
        local t = thickness:GetValue()

        if outline_cb:GetValue() then
            draw.Color(0, 0, 0, a)
            DrawCrosshairShape(cx + 1, cy, s, t)
            DrawCrosshairShape(cx - 1, cy, s, t)
            DrawCrosshairShape(cx, cy + 1, s, t)
            DrawCrosshairShape(cx, cy - 1, s, t)
        end

        draw.Color(r, g, b, a)
        DrawCrosshairShape(cx, cy, s, t)
    end
end)

callbacks.Register("Unload", function()
    client.Command("crosshair 1", false)
    print(" oyun crossunu gotune sok.")
end)

--defib (sag ustte olcak ona gore)
local menu = gui.Tab(gui.Reference("Visuals"), "defib_tab", "Defib Display")
local defib_checkbox = gui.Checkbox(menu, "defib_display_enabled", "Defib Timer Göster", true)

local active = false
local endTime = 0
local status = ""
local lastUpdate = 0

client.AllowListener("defibrillator_begin")
client.AllowListener("defibrillator_used")
client.AllowListener("defibrillator_interrupted")
client.AllowListener("defibrillator_used_fail")

callbacks.Register("FireGameEvent", function(e)
    if not defib_checkbox:GetValue() then return end

    local name = e:GetName()
    if name == "defibrillator_begin" then
        active = true
        endTime = globals.RealTime() + 3
        status = "Defib basladi"
        lastUpdate = globals.RealTime()

    elseif name == "defibrillator_interrupted" then
        active = false
        status = "Defib iptal"
        lastUpdate = globals.RealTime()

    elseif name == "defibrillator_used_fail" then
        active = false
        status = "Defib fail"
        lastUpdate = globals.RealTime()

    elseif name == "defibrillator_used" then
        active = false
        status = "Defib tamamlandi"
        lastUpdate = globals.RealTime()
    end
end)

callbacks.Register("Draw", function()
    if not defib_checkbox:GetValue() then return end

    local sw, sh = draw.GetScreenSize()

    local drawTime = 5 
    if status ~= "" and globals.RealTime() - lastUpdate < drawTime then
        local text = status
        if active then
            local timeLeft = math.max(0, endTime - globals.RealTime())
            text = status .. " | " .. string.format("%.1f", timeLeft)
        end

        local tw, th = draw.GetTextSize(text)
        local x, y = sw - tw - 20, 20

        draw.Color(0, 0, 0, 150)
        draw.FilledRect(x - 5, y - 5, x + tw + 5, y + th + 5)

        draw.Color(255, 255, 255, 255)
        draw.Text(x, y, text)
    end
end)

local ref    = gui.Reference("Misc", "General", "Extra")
local enable = gui.Checkbox(ref, "ffi.chams.on", "XQZ Chams", false)
local col_r  = gui.Slider(ref, "ffi.chams.r", "Chams R", 255, 0, 255, 1)
local col_g  = gui.Slider(ref, "ffi.chams.g", "Chams G",  80, 0, 255, 1)
local col_b  = gui.Slider(ref, "ffi.chams.b", "Chams B",  80, 0, 255, 1)

ffi.cdef[[
    void* GetModuleHandleA(const char* lpModuleName);
    void* GetProcAddress(void* hModule, const char* lpProcName);
    int   VirtualProtect(void* lpAddress, size_t dwSize, unsigned long flNewProtect, unsigned long* lpflOldProtect);
]]

local function GetInterface(dll, name)
    local mod = ffi.C.GetModuleHandleA(dll)
    if mod == nil then return nil end
    local ci = ffi.cast("void*(__cdecl*)(const char*, int*)",
        ffi.C.GetProcAddress(mod, "CreateInterface"))
    return ci(name, nil)
end

local model_render = GetInterface("engine.dll", "VEngineModel016")
if not model_render then print("[Chams] model_render yok") return end

local mat_sys = GetInterface("materialsystem.dll", "VMaterialSystem080")
if not mat_sys then print("[Chams] mat_sys yok") return end


local mat = ffi.cast("void*(__thiscall*)(void*, const char*, const char*, int, const char*)",
    ffi.cast("uintptr_t**", mat_sys)[0][71])(mat_sys, "debug/debugambientcube", "Model textures", 0, nil)

if mat == nil or ffi.cast("uintptr_t", mat) == 0 then print("[Chams] mat yok") return end


ffi.cast("void(__thiscall*)(void*)", ffi.cast("uintptr_t**", mat)[0][12])(mat)


local set_flag  = ffi.cast("void(__thiscall*)(void*, int, int)",            ffi.cast("uintptr_t**", mat)[0][29])
local set_color = ffi.cast("void(__thiscall*)(void*, float, float, float)", ffi.cast("uintptr_t**", mat)[0][28])
local override  = ffi.cast("void(__thiscall*)(void*, void*, int)",          ffi.cast("uintptr_t**", model_render)[0][1])

local IGNOREZ = bit.lshift(1, 15); local ZNEARER = bit.lshift(1, 10)
local NOCULL  = bit.lshift(1, 13); local NOFOG   = bit.lshift(1, 14)


local DME_t    = ffi.typeof("void(__fastcall*)(void*, void*, void*, void*, void*)")
local vtable   = ffi.cast("uintptr_t**", model_render)[0]
local slot_ptr = vtable + 19
local orig     = slot_ptr[0]
local GC       = {}
local dbg_cnt  = 0

print("[Chams] DrawModelExecute orig @ " .. tostring(orig))

local function Hooked_DME(ecx, edx, state, info, btw)
    if not enable:GetValue() then
        ffi.cast(DME_t, orig)(ecx, edx, state, info, btw)
        return
    end

    dbg_cnt = dbg_cnt + 1
    if dbg_cnt <= 3 then print("[Chams] Hook calisti! cnt=" .. dbg_cnt) end

  
    local eidx = ffi.cast("int*", ffi.cast("uintptr_t", info) + 48)[0]

   
    if eidx < 1 or eidx > 32 then
        ffi.cast(DME_t, orig)(ecx, edx, state, info, btw)
        return
    end

    local r = col_r:GetValue() / 255.0
    local g = col_g:GetValue() / 255.0
    local b = col_b:GetValue() / 255.0

   
    set_flag(mat, IGNOREZ, 1); set_flag(mat, ZNEARER, 1)
    set_flag(mat, NOCULL, 1);  set_flag(mat, NOFOG, 1)
    set_color(mat, r * 0.4, g * 0.4, b * 0.4)
    override(model_render, mat, 0)
    ffi.cast(DME_t, orig)(ecx, edx, state, info, btw)
    override(model_render, nil, 0)

   
    set_flag(mat, IGNOREZ, 0)
    set_color(mat, r, g, b)
    override(model_render, mat, 0)
    ffi.cast(DME_t, orig)(ecx, edx, state, info, btw)
    override(model_render, nil, 0)
end

GC.fn = ffi.cast(DME_t, Hooked_DME)

local old = ffi.new("unsigned long[1]")
ffi.C.VirtualProtect(ffi.cast("void*", slot_ptr), 4, 0x40, old)
slot_ptr[0] = ffi.cast("uintptr_t", GC.fn)
ffi.C.VirtualProtect(ffi.cast("void*", slot_ptr), 4, old[0], old)

callbacks.Register("Unload", "ffichams", function()
    local old2 = ffi.new("unsigned long[1]")
    ffi.C.VirtualProtect(ffi.cast("void*", slot_ptr), 4, 0x40, old2)
    slot_ptr[0] = orig
    ffi.C.VirtualProtect(ffi.cast("void*", slot_ptr), 4, old2[0], old2)
    override(model_render, nil, 0)
    print("[Chams] Temizlendi")
end)
