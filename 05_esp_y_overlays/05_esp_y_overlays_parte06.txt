    if not me or not me:IsAlive() then return end
    local mypos  = me:GetAbsOrigin()
    local max_m  = max_dist:GetValue()
    local r, g, b, a = col:GetValue()

    local cabs = entities.FindByClass("CDynamicProp") or {}
    for _, e in pairs(cabs) do
        local ok, name = pcall(function() return e:GetName() end)
        if not ok or not name then goto continue end
        if not (name:lower():find("medical") or name:lower():find("medkit")) then goto continue end

        local ok2, pos = pcall(function() return e:GetAbsOrigin() end)
        if not ok2 or not pos then goto continue end

        local dist = (pos - mypos):Length() * 0.01905
        if dist > max_m then goto continue end

        local sx, sy = client.WorldToScreen(pos + Vector3(0, 0, 40))
        if not sx or not sy then goto continue end

        draw.SetFont(font)
        local txt = "Health Cabinet"

        draw.Color(0, 0, 0, 200)
        draw.Text(sx - tw/2 + 1, sy - th/2 + 1, txt)
        draw.Color(r, g, b, a)
        draw.Text(sx - tw/2, sy - th/2, txt)

local ref    = gui.Reference("Visuals", "Other")
local gb     = gui.Groupbox(ref, "Car Alarm", 325, 315, 300, 0)
local enable = gui.Checkbox(gb, "ca.enable", "Enable",         false)
local dur    = gui.Slider(gb,   "ca.dur",    "Gösterim Süresi (s)", 5, 2, 15)
local col    = gui.ColorPicker(gb, "ca.col", "Renk", 255, 180, 0, 255)

local font_big = draw.CreateFont("Tahoma", 15, 700)
local font_sm  = draw.CreateFont("Tahoma", 12, 600)

local alarms = {}

client.AllowListener("triggered_car_alarm")

callbacks.Register("FireGameEvent", "ca_ev", function(event)
    if not enable:GetValue() then return end
    if event:GetName() ~= "triggered_car_alarm" then return end

    local userid = event:GetInt("userid")
    local pIdx   = client.GetPlayerIndexByUserID(userid)
    local pName  = "?"
    local pTeam  = 0

    if pIdx and pIdx > 0 then
        local ent = entities.GetByIndex(pIdx)
        if ent then
            pName = ent:GetName() or "?"
            pTeam = ent:GetTeamNumber()
        end
    end

    table.insert(alarms, 1, {
        time  = globals.RealTime(),
        name  = pName,
        team  = pTeam,
    })

    if #alarms > 5 then table.remove(alarms) end

    print(string.format("[Car Alarm] %s tetikledi", pName))
end)

callbacks.Register("Draw", "ca_draw", function()
    if not enable:GetValue() then return end
    if #alarms == 0 then return end

    local now  = globals.RealTime()
    local d    = dur:GetValue()
    local r, g, b, _ = col:GetValue()
    local sw, sh = draw.GetScreenSize()

    local i = 1
    while i <= #alarms do
        if now - alarms[i].time > d then
            table.remove(alarms, i)
        else
            i = i + 1
        end
    end

    if #alarms == 0 then return end

    local pad = 8
    local lh  = 18
    local w   = 260
    local h   = pad*2 + 20 + #alarms * lh
    local px  = sw/2 - w/2
    local py  = sh * 0.2

    draw.Color(0, 0, 0, 170)
    draw.FilledRect(px, py, px+w, py+h)
    draw.Color(r, g, b, 200)
    draw.OutlinedRect(px, py, px+w, py+h)

    draw.SetFont(font_big)
    local title = "ARABA ALARMI"
    local tw, _ = draw.GetTextSize(title)
    draw.Color(0, 0, 0, 180)
    draw.Text(px+w/2-tw/2+1, py+pad+1, title)
    draw.Color(r, g, b, 255)
    draw.Text(px+w/2-tw/2, py+pad, title)

    draw.SetFont(font_sm)
    local ly = py + pad + 20

    for _, alarm in ipairs(alarms) do
        local age   = now - alarm.time
        local ratio = math.max(0, 1 - age/d)
        local alpha = math.floor(ratio * 255)

        local tr, tg, tb
        if alarm.team == 2 then
            tr, tg, tb = 80, 180, 255 
        elseif alarm.team == 3 then
            tr, tg, tb = 255, 80, 80    
        else
            tr, tg, tb = 200, 200, 200
        end

        local ago  = string.format("%.0fs önce", age)
        local txt  = alarm.name .. " — " .. ago
        local tw2, _ = draw.GetTextSize(txt)

        draw.Color(0, 0, 0, math.floor(alpha*0.6))
        draw.Text(px+pad+1, ly+1, txt)
        draw.Color(tr, tg, tb, alpha)
        draw.Text(px+pad, ly, txt)

        ly = ly + lh
    end
end)

callbacks.Register("Unload", function()
    alarms = {}
end)

local ref    = gui.Reference("Visuals", "World")
local gb     = gui.Groupbox(ref, "Charger ESP", 325, 315, 300, 0)

local enable = gui.Checkbox(gb, "chg.enable",   "Enable",           false)
local show_path = gui.Checkbox(gb, "chg.path",  "Charge Path",      true)
local show_cd   = gui.Checkbox(gb, "chg.cd",    "Cooldown Bar",     true)
local show_tgt  = gui.Checkbox(gb, "chg.tgt",   "Show Target",      true)

local col    = gui.ColorPicker(gb, "chg.col",   "Color", 255, 120, 0, 255)
local font   = draw.CreateFont("Tahoma", 12, 700)

local CHARGE_DURATION = 3.0
local COOLDOWN_TIME   = 12.2

local charge_cache = {} -- idx → { was_charging, end_pos }
local cd_cache     = {}

local function T(x, y, t, r, g, b)
    draw.SetFont(font)
    draw.Color(0,0,0,200) draw.Text(x+1,y+1,t)
    draw.Color(r,g,b,255) draw.Text(x,y,t)
end

local function DrawLine3D(p1, p2, r, g, b, a)
    local sx1, sy1 = client.WorldToScreen(p1)
    local sx2, sy2 = client.WorldToScreen(p2)
    if not sx1 or not sx2 then return end
    draw.Color(0,0,0,math.floor(a*0.4)) draw.Line(sx1+1,sy1+1,sx2+1,sy2+1)
    draw.Color(r,g,b,a) draw.Line(sx1,sy1,sx2,sy2)
end

local function DrawCircle3D(origin, radius, r, g, b, a)
    local px, py = nil, nil
    for i = 0, 32 do
        local ang = (i/32) * math.pi * 2
        local wx = origin.x + math.cos(ang) * radius
        local wy = origin.y + math.sin(ang) * radius
        local sx, sy = client.WorldToScreen(Vector3(wx, wy, origin.z + 5))
        if sx and sy then
            if px then
                draw.Color(r,g,b,a)
                draw.Line(px,py,sx,sy)
            end
            px, py = sx, sy
        end
    end
end

callbacks.Register("Draw", function()
    if not enable:GetValue() then return end

    local lp = entities.GetLocalPlayer()
    if not lp then return end

    local now = globals.CurTime()
    local r,g,b,a = col:GetValue()
    local lp_pos = lp:GetAbsOrigin()

    for _, ent in pairs(entities.FindByClass("CTerrorPlayer") or {}) do
        if not ent:IsAlive() or ent:IsDormant() then goto skip end
        if ent:GetTeamNumber() ~= 3 then goto skip end
        if (ent:GetPropInt("m_zombieClass") or 0) ~= 6 then goto skip end

        local idx = ent:GetIndex()
        local pos = ent:GetAbsOrigin()
        if not pos then goto skip end

        local sx, sy = client.WorldToScreen(pos + Vector3(0,0,80))

        local vx = ent:GetPropFloat("localdata", "m_vecVelocity[0]") or 0
        local vy = ent:GetPropFloat("localdata", "m_vecVelocity[1]") or 0
        local spd = math.sqrt(vx*vx + vy*vy)

        local isCharging = spd > 300
        local wasCharging = charge_cache[idx] and charge_cache[idx].was_charging or false

        if not charge_cache[idx] then
            charge_cache[idx] = {}
        end

        if isCharging then
            charge_cache[idx].was_charging = true

            if not charge_cache[idx].end_pos then
                if spd > 0 then
                    local dir_x = vx / spd
                    local dir_y = vy / spd

                   local CHARGE_MULTIPLIER = 1.25
                   local max_dist = spd * CHARGE_DURATION * CHARGE_MULTIPLIER

                    local start_pos = pos + Vector3(0,0,40)
                    local max_end = Vector3(
                        pos.x + dir_x * max_dist,
                        pos.y + dir_y * max_dist,
                        pos.z + 40
                    )

                    local tr = engine.TraceLine(start_pos, max_end, 0x46004003)

                    local dist = max_dist
                    if tr and tr.fraction < 1 then
                        dist = max_dist * tr.fraction
                    end

                    charge_cache[idx].end_pos = Vector3(
                        pos.x + dir_x * dist,
                        pos.y + dir_y * dist,
                        pos.z
                    )
                end
            end

            local end_pos = charge_cache[idx].end_pos

            if end_pos and show_path:GetValue() then
                DrawLine3D(pos + Vector3(0,0,40), end_pos + Vector3(0,0,40), 255,50,50,220)
                DrawCircle3D(end_pos, 40, 255,50,50,150)

                local ex, ey = client.WorldToScreen(end_pos + Vector3(0,0,40))
                if ex and ey then
                    draw.Color(255,50,50,200)
                    draw.FilledCircle(ex,ey,5)

                    local d = (end_pos - pos):Length()
                    T(ex-15, ey-16, string.format("%.0fm", d*0.01905), 255,50,50)
                end
            end

            if sx and sy then
                T(sx-25, sy-30, "CHARGING", 255,50,50)
            end

        else
            if wasCharging then
                cd_cache[idx] = now + COOLDOWN_TIME
            end

            charge_cache[idx].was_charging = false
            charge_cache[idx].end_pos = nil

            if show_cd:GetValue() and sx and sy then
                local cd_end = cd_cache[idx] or 0
                local cd_remain = math.max(0, cd_end - now)
                local ratio = cd_remain / COOLDOWN_TIME

                local barW, barH = 70, 6

                draw.FilledRect(bx,by,bx + math.floor(barW*ratio),by+barH)
