local function GetInterface(dll, name)
    local module = ffi.C.GetModuleHandleA(dll)
    if module == nil then return nil end
    local ci_addr = ffi.C.GetProcAddress(module, "CreateInterface")
    if ci_addr == nil then return nil end
    local ci = ffi.cast("void*(__cdecl*)(const char*, int*)", ci_addr)
    return ci(name, nil)
end

local first_mat = ffi.cast(
    "uint16_t(__thiscall*)(void*)",
    ffi.cast("uintptr_t**", mat_system)[0][72]
)
local next_mat = ffi.cast(
    "uint16_t(__thiscall*)(void*, uint16_t)",
    ffi.cast("uintptr_t**", mat_system)[0][73]
)
local invalid_mat = ffi.cast(
    "uint16_t(__thiscall*)(void*)",
    ffi.cast("uintptr_t**", mat_system)[0][74]
)
local get_mat = ffi.cast(
    "void*(__thiscall*)(void*, uint16_t)",
    ffi.cast("uintptr_t**", mat_system)[0][75]
)

local TARGETS = {
    "vgui_white",
    "800corner1",
    "800corner2",
    "800corner3",
    "800corner4",
}

local s_mats      = {}
local s_found     = false
local s_frame_delay = 0

local function GetMatName(mat)
    local ok, name = pcall(function()
        local vt = ffi.cast("uintptr_t**", mat)[0]
        local fn = ffi.cast("const char*(__thiscall*)(void*)", vt[0])
        local raw = fn(mat)
        if raw == nil then return nil end
        return ffi.string(raw)
    end)
    if not ok then return nil end
    return name
end

local function ColorModulate(mat, r, g, b)
    pcall(function()
        local vt = ffi.cast("uintptr_t**", mat)[0]
        local fn = ffi.cast("void(__thiscall*)(void*, float, float, float)", vt[28])
        fn(mat, r, g, b)
    end)
end

local function AlphaModulate(mat, a)
    pcall(function()
        local vt = ffi.cast("uintptr_t**", mat)[0]
        local fn = ffi.cast("void(__thiscall*)(void*, float)", vt[27])
        fn(mat, a)
    end)
end

local function FindMaterials()
    local ok_iter = pcall(function()
        local h       = first_mat(mat_system)
        local invalid = invalid_mat(mat_system)
        local count   = 0

        while h ~= invalid do
            local ok_get, mat = pcall(get_mat, mat_system, h)
            if ok_get and mat ~= nil and ffi.cast("uintptr_t", mat) ~= 0 then
                local name = GetMatName(mat)
                if name then
                    for i, target in ipairs(TARGETS) do
                        if not s_mats[i] and name:find(target, 1, true) then
                            s_mats[i] = mat
                            print("[Con] Bulundu: " .. name)
                        end
                    end
                end
            end

            local all_found = true
            for i = 1, #TARGETS do
                if not s_mats[i] then all_found = false break end
            end
            if all_found then
                s_found = true
                print("[Con] Tum materyaller bulundu!")
                break
            end

            count = count + 1
            if count > 10000 then break end

            local ok_next, next_h = pcall(next_mat, mat_system, h)
            if not ok_next then break end
            h = next_h
        end
    end)
end

local function ResetColors()
    for _, mat in ipairs(s_mats) do
        if mat then
            ColorModulate(mat, 1.0, 1.0, 1.0)
            AlphaModulate(mat, 1.0)
        end
    end
end

callbacks.Register("Draw", "console_color_draw", function()
    if s_frame_delay < 300 then
        s_frame_delay = s_frame_delay + 1
        return
    end

    if not s_found then
        FindMaterials()
        return
    end

    if not con_enable:GetValue() then
        ResetColors()
        return
    end

    local r, g, b = con_color:GetValue()
    local rf, gf, bf = r/255.0, g/255.0, b/255.0

    for _, mat in ipairs(s_mats) do
        if mat then
            ColorModulate(mat, rf, gf, bf)
            AlphaModulate(mat, 1.0)
        end
    end
end)

callbacks.Register("Unload", function()
    ResetColors()
    s_mats        = {}
    s_found       = false
    s_frame_delay = 0
end)

local find_var_fn = ffi.cast("void*(__thiscall*)(void*, const char*)", ffi.cast("uintptr_t**", icvar)[0][13])

local function SetupConVar(name)
    local ptr = find_var_fn(icvar, name)
    if ptr == nil or ffi.cast("uintptr_t", ptr) == 0 then return nil end
    
    return {
        ptr = ptr,
        SetFloat  = ffi.cast("void(__thiscall*)(void*, float)", ffi.cast("uintptr_t**", ptr)[0][13]),
        RemoveFlags = ffi.cast("void(__thiscall*)(void*, int)", ffi.cast("uintptr_t**", ptr)[0][4])
    }
end

local fontCache = nil
local lastFontState = ""

local function UpdateFont()
    local state =
        font_select:GetValue() .. "|" ..
        font_size:GetValue() .. "|" ..
        tostring(font_bold:GetValue())

    if state ~= lastFontState then
        local fonts = { "Tahoma", "Verdana", "Arial", "Trebuchet MS" }
        local weight = font_bold:GetValue() and 800 or 400
        fontCache = draw.CreateFont(fonts[font_select:GetValue()+1], font_size:GetValue(), weight)
        lastFontState = state
    end
end

local function SafeGetProp(ent, prop)
    local ok, val = pcall(function() return ent:GetProp(prop) end)
    if not ok then return nil end
    return val
end

local function IsGhost(ent)
    local g = SafeGetProp(ent, "m_isGhost")
    return (g == 1 or g == true)
end

local function IsDormantSafe(ent)
    local ok, val = pcall(function() return ent:IsDormant() end)
    if not ok then return false end
    return val == true
end

local function GetPlayers()
    local p = entities.FindByClass("CTerrorPlayer")
    if not p or #p == 0 then
        p = entities.FindByClass("CCSPlayer")
    end
    return p
end

local function IsEntityDeadSafe(ent)
    if not ent then return false end
    local ok, alive = pcall(function() return ent:IsAlive() end)
    if not ok then return true end
    return alive == false
end

local function GetAimTarget(me, maxDist)
    local myPos = me:GetAbsOrigin() + Vector3(0,0,60)
    local fwd = engine.GetViewAngles():Forward()
    local best, bestDot = nil, 0.985
    local myTeam = me:GetTeamNumber()

    for _, ent in pairs(GetPlayers() or {}) do
        if ent ~= me and ent:IsAlive() and not IsDormantSafe(ent) then
            local tTeam = ent:GetTeamNumber()
            if myTeam == 2 then
                if tTeam ~= 3 or IsGhost(ent) then goto cont end
            elseif myTeam == 3 then
                if tTeam ~= 2 then goto cont end
            end

            local pos = ent:GetAbsOrigin() + Vector3(0,0,60)
            local dir = pos - myPos
            local dist = dir:Length()
            if dist < maxDist then
                dir = dir / dist
                local dot = fwd:Dot(dir)
                if dot > bestDot then
                    bestDot = dot
                    best = ent
                end
            end
        end
        ::cont::
    end
    return best
end

local effects = {}
local lastTarget = nil

callbacks.Register("Draw", function()

    if not enable:GetValue() then return end

    UpdateFont()
    draw.SetFont(fontCache)

    local me = entities.GetLocalPlayer()
    if not me or not me:IsAlive() then
        lastTarget = nil
        return
    end

    local target = GetAimTarget(me, 1500)

    if lastTarget and IsEntityDeadSafe(lastTarget) then
        local pos = lastTarget:GetAbsOrigin()
        effects[#effects+1] = {
            pos = Vector3(pos.x, pos.y, pos.z),
            start = globals.RealTime(),
            expire = globals.RealTime() + duration:GetValue()
        }
        lastTarget = nil
    end

    if target and target:IsAlive() then
        lastTarget = target
    end

    local r,g,b,a = color:GetValue()
    local txt = text_input:GetValue()

    for i = #effects, 1, -1 do
        local e = effects[i]
        local t = globals.RealTime()

        if t > e.expire then
            table.remove(effects, i)
        else
            local prog = t - e.start
            local life = duration:GetValue()
            local worldPos = e.pos + Vector3(0,0,70 + prog * 15)
            local sx, sy = client.WorldToScreen(worldPos)

            if sx and sy then
                local alpha = math.floor((1 - prog / life) * a)
                draw.Color(r,g,b,alpha)
                draw.Text(sx, sy, txt)
            end
        end
    end
end)
