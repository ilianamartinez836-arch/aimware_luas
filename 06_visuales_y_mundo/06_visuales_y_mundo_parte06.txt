local ref = gui.Reference("Misc", "General", "Extra")
local lowres_enable = gui.Checkbox(ref, "misc.visuals.mat_showlowresimage", "Low-Res", false)

ffi.cdef[[
    void* GetModuleHandleA(const char* lpModuleName);
    void* GetProcAddress(void* hModule, const char* lpProcName);
]]

local FCVAR_CHEAT = 16384

local function GetInterface(dll, name)
    local module = ffi.C.GetModuleHandleA(dll)
    if module == nil then return nil end
    local ci_addr = ffi.C.GetProcAddress(module, "CreateInterface")
    if ci_addr == nil then return nil end
    local ci = ffi.cast("void*(__cdecl*)(const char*, int*)", ci_addr)
    return ci(name, nil)
end

local icvar = GetInterface("vstdlib.dll", "VEngineCvar007")
if not icvar then return end

local find_var_fn = ffi.cast("void*(__thiscall*)(void*, const char*)", ffi.cast("uintptr_t**", icvar)[0][13])

local function SetupConVar(name)
    local ptr = find_var_fn(icvar, name)
    if ptr == nil or ffi.cast("uintptr_t", ptr) == 0 then return nil end
    
    return {
        ptr = ptr,
        SetFloat  = ffi.cast("void(__thiscall*)(void*, float)", ffi.cast("uintptr_t**", ptr)[0][13]),
        RemoveFlags = ffi.cast("void(__thiscall*)(void*, int)", ffi.cast("uintptr_t**", ptr)[0][4])
    }
end

local cv_lowres = SetupConVar("mat_showlowresimage")
if cv_lowres then cv_lowres.RemoveFlags(cv_lowres.ptr, FCVAR_CHEAT) end

local last_state = false

callbacks.Register("CreateMove", "mat_lowres_logic", function(cmd)
    local enabled = lowres_enable:GetValue()
    
    if enabled == last_state then return end
    last_state = enabled

    if enabled then
        if cv_lowres then cv_lowres.SetFloat(cv_lowres.ptr, 1.0) end 
    else
        if cv_lowres then cv_lowres.SetFloat(cv_lowres.ptr, 0.0) end 
    end
end)

callbacks.Register("Unload", function()
    if cv_lowres then cv_lowres.SetFloat(cv_lowres.ptr, 0.0) end
end)

===== FILE: Model Color Changer =====
local ref = gui.Reference("Misc", "General", "Extra")

local activeBox   = gui.Checkbox(ref, "custom_tint_active", "Change Model Coloe", false)
local rainbowBox  = gui.Checkbox(ref, "custom_tint_rainbow", "Rainbow Mode", false)
local speedSlider = gui.Slider(ref, "custom_tint_speed", "Rainbow Speed", 2.0, 0.1, 10.0, 0.1)

local rSlider     = gui.Slider(ref, "custom_tint_r", "Kirmizi (R)", 0, 0, 255, 1)
local gSlider     = gui.Slider(ref, "custom_tint_g", "Yesil (G)", 180, 0, 255, 1)
local bSlider     = gui.Slider(ref, "custom_tint_b", "Mavi (B)", 255, 0, 255, 1)

local wasActive = false

local function hsvToRgb(h, s, v)
    local r, g, b
    local i = math.floor(h * 6)
    local f = h * 6 - i
    local p = v * (1 - s)
    local q = v * (1 - f * s)
    local t = v * (1 - (1 - f) * s)

    local rem = i % 6
    if rem == 0 then r, g, b = v, t, p
    elseif rem == 1 then r, g, b = q, v, p
    elseif rem == 2 then r, g, b = p, v, t
    elseif rem == 3 then r, g, b = p, q, v
    elseif rem == 4 then r, g, b = t, p, v
    elseif rem == 5 then r, g, b = v, p, q
    end

    return math.floor(r * 255), math.floor(g * 255), math.floor(b * 255)
end

callbacks.Register("Draw", "l4d2_simple_color_tint", function()
    local pLocal = entities.GetLocalPlayer()
    if not pLocal or not pLocal:IsAlive() then return end

    if activeBox:GetValue() then
        wasActive = true

        local r, g, b

        if rainbowBox:GetValue() then
            local speed = speedSlider:GetValue()
            local hue = (globals.RealTime() * speed * 0.1) % 1.0
            r, g, b = hsvToRgb(hue, 1.0, 1.0)
        else
            r = math.floor(rSlider:GetValue())
            g = math.floor(gSlider:GetValue())
            b = math.floor(bSlider:GetValue())
        end

        local clrInt = r + (g * 256) + (b * 65536)
        pLocal:SetPropInt(clrInt, "m_clrRender")
    else
        if wasActive then
            pLocal:SetPropInt(16777215, "m_clrRender")
            wasActive = false
        end
    end
end)


===== END: Model Color Changer =====

local ref = gui.Reference("Misc", "General", "Extra")


local enableBox   = gui.Checkbox(ref, "mscale_master", "Master Model Scaler", false)

local localSlider = gui.Slider(ref, "mscale_local", "Local", 1.0, 0.2, 3.0, 0.1)
local survSlider  = gui.Slider(ref, "mscale_surv", "Survivors", 1.0, 0.2, 3.0, 0.1)
local siSlider    = gui.Slider(ref, "mscale_si", "Special Infecteds", 1.0, 0.2, 3.0, 0.1)
local ciSlider    = gui.Slider(ref, "mscale_ci", "Commons", 1.0, 0.2, 3.0, 0.1)

local wasEnabled = false

callbacks.Register("Draw", "l4d2_multi_entity_scale", function()
    local isEnabled = enableBox:GetValue()

  
    if not isEnabled then
        if wasEnabled then
            local pLocal = entities.GetLocalPlayer()
            if pLocal then pLocal:SetPropFloat(1.0, "m_flModelScale") end

            local players = entities.FindByClass("CTerrorPlayer")
            for _, ent in pairs(players) do
                ent:SetPropFloat(1.0, "m_flModelScale")
            end

            local infected = entities.FindByClass("Infected")
            for _, ent in pairs(infected) do
                ent:SetPropFloat(1.0, "m_flModelScale")
            end

            wasEnabled = false
        end
        return
    end

    wasEnabled = true

    local pLocal = entities.GetLocalPlayer()
    local localScale = localSlider:GetValue()
    local survScale  = survSlider:GetValue()
    local siScale    = siSlider:GetValue()
    local ciScale    = ciSlider:GetValue()

    
    if pLocal and pLocal:IsAlive() then
        pLocal:SetPropFloat(localScale, "m_flModelScale")
    end

    
    local players = entities.FindByClass("CTerrorPlayer")
    for _, ent in pairs(players) do
        if ent and ent:IsAlive() and ent:GetIndex() ~= (pLocal and pLocal:GetIndex()) then
            local team = ent:GetTeamNumber()
            if team == 2 then
           
                ent:SetPropFloat(survScale, "m_flModelScale")
            elseif team == 3 then
              
                ent:SetPropFloat(siScale, "m_flModelScale")
            end
        end
    end

 
    local infected = entities.FindByClass("Infected")
    for _, ent in pairs(infected) do
        if ent and ent:IsAlive() then
            ent:SetPropFloat(ciScale, "m_flModelScale")
        end
    end
end)


===== END: Model Scale Changer =====
