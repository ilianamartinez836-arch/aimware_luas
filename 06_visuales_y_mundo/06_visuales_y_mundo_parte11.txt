ffi.cdef[[
    void* GetModuleHandleA(const char* lpModuleName);
    void* GetProcAddress(void* hModule, const char* lpProcName);
    int   VirtualProtect(void* lpAddress, size_t dwSize, uint32_t flNewProtect, uint32_t* lpflOldProtect);
]]

local FCVAR_CHEAT = 16384

local engine_module = ffi.cast("uintptr_t", ffi.C.GetModuleHandleA("engine.dll"))
local client_module = ffi.cast("uintptr_t", ffi.C.GetModuleHandleA("client.dll"))

if engine_module == 0 then print("[Sky] engine.dll none") return end
if client_module == 0 then print("[Sky] client.dll none") return end

print(string.format("[Sky] engine.dll: 0x%X", tonumber(engine_module)))
print(string.format("[Sky] client.dll: 0x%X", tonumber(client_module)))


local function GetInterface(dll, name)
    local module = ffi.C.GetModuleHandleA(dll)
    if module == nil then return nil end
    local ci_addr = ffi.C.GetProcAddress(module, "CreateInterface")
    if ci_addr == nil then return nil end
    local ci = ffi.cast("void*(__cdecl*)(const char*, int*)", ci_addr)
    return ci(name, nil)
end

local icvar = GetInterface("vstdlib.dll", "VEngineCvar007")
if icvar == nil then print("[Sky] ICvar bulunamadi") return end

local find_var_fn = ffi.cast(
    "void*(__thiscall*)(void*, const char*)",
    ffi.cast("uintptr_t**", icvar)[0][13]
)

local function FindConVar(name)
    local ok, result = pcall(function()
        return find_var_fn(icvar, name)
    end)
    if not ok or result == nil then return nil end
    if ffi.cast("uintptr_t", result) == 0 then return nil end
    return result
end

local function RemoveCheatFlag(cv)
    if cv == nil then return end
    local fn = ffi.cast(
        "void(__thiscall*)(void*, int)",
        ffi.cast("uintptr_t**", cv)[0][4]
    )
    pcall(fn, cv, FCVAR_CHEAT)
end

local function SetConVarString(cv, value)
    if cv == nil then return end
    local fn = ffi.cast(
        "void(__thiscall*)(void*, const char*)",
        ffi.cast("uintptr_t**", cv)[0][12]
    )
    local buf = ffi.new("char[256]")
    ffi.copy(buf, value)
    pcall(fn, cv, buf)
end

local cv_sv_skyname = FindConVar("sv_skyname")
if cv_sv_skyname then
    RemoveCheatFlag(cv_sv_skyname)
    print("[Sky] fucksky found")
else
    print("[Sky] none")
end


local r_loadskys_addr    = ffi.cast("void*", engine_module + 868112)
local skyname_strval_ptr = client_module + 0x7806A8 + 0x24

print(string.format("[Sky] R_LoadSkys: 0x%X",     tonumber(ffi.cast("uintptr_t", r_loadskys_addr))))
print(string.format("[Sky] strval ptr: 0x%X",     tonumber(skyname_strval_ptr)))

local r_loadskys_fn = ffi.cast("void(__cdecl*)()", r_loadskys_addr)
local sky_buf       = ffi.new("char[256]")

local function ApplySky(name)
    -- cvar set shit
    if cv_sv_skyname then
        SetConVarString(cv_sv_skyname, name)
    end

    
    ffi.copy(sky_buf, name)
    local original_str = ffi.cast("uintptr_t*", skyname_strval_ptr)[0]
    ffi.cast("uintptr_t*", skyname_strval_ptr)[0] = ffi.cast("uintptr_t", sky_buf)

    -- call
    local ok, err = pcall(r_loadskys_fn)

    
    ffi.cast("uintptr_t*", skyname_strval_ptr)[0] = original_str

    if not ok then
        print("[Sky] R_LoadSkys shitto errorito: " .. tostring(err))
    else
        print("[Sky] Sky changed: " .. name)
    end
end

local last_idx   = -1
local last_state = false

callbacks.Register("CreateMove", "sky_createmove", function(cmd)
    local enabled = skybox_enable:GetValue()
    local idx     = skybox_list:GetValue()

    if enabled ~= last_state or (enabled and idx ~= last_idx) then
        last_state = enabled
        last_idx   = idx

        if enabled then
            ApplySky(SKYBOXES[idx])
        else
            ApplySky(DEFAULT_SKY)
        end
    end
end)

callbacks.Register("Unload", function()
    ApplySky(DEFAULT_SKY)
end)

print("skybox changer ready")

local ref = gui.Reference("Misc", "General", "Extra")
local colorprops_enable = gui.Checkbox(ref, "misc.visuals.neon_props", "Static Props Color", false)

ffi.cdef[[
    void* GetModuleHandleA(const char* lpModuleName);
    void* GetProcAddress(void* hModule, const char* lpProcName);
]]

local FCVAR_CHEAT = 16384

local function GetInterface(dll, name)
    local module = ffi.C.GetModuleHandleA(dll)
    if module == nil then return nil end
    local ci_addr = ffi.C.GetProcAddress(module, "CreateInterface")
    if ci_addr == nil then return nil end
    local ci = ffi.cast("void*(__cdecl*)(const char*, int*)", ci_addr)
    return ci(name, nil)
end

local icvar = GetInterface("vstdlib.dll", "VEngineCvar007")
if not icvar then return end

local find_var_fn = ffi.cast("void*(__thiscall*)(void*, const char*)", ffi.cast("uintptr_t**", icvar)[0][13])

local function SetupConVar(name)
    local ptr = find_var_fn(icvar, name)
    if ptr == nil or ffi.cast("uintptr_t", ptr) == 0 then return nil end
    
    return {
        ptr = ptr,
        SetFloat  = ffi.cast("void(__thiscall*)(void*, float)", ffi.cast("uintptr_t**", ptr)[0][13]),
        RemoveFlags = ffi.cast("void(__thiscall*)(void*, int)", ffi.cast("uintptr_t**", ptr)[0][4])
    }
end

local cv_props = SetupConVar("r_colorstaticprops")
if cv_props then cv_props.RemoveFlags(cv_props.ptr, FCVAR_CHEAT) end

local last_state = false

callbacks.Register("CreateMove", "r_colorprops_logic", function(cmd)
    local enabled = colorprops_enable:GetValue()
    
    if enabled == last_state then return end
    last_state = enabled

    if enabled then
        if cv_props then cv_props.SetFloat(cv_props.ptr, 1.0) end 
    else
        if cv_props then cv_props.SetFloat(cv_props.ptr, 0.0) end 
    end
end)

callbacks.Register("Unload", function()
    if cv_props then cv_props.SetFloat(cv_props.ptr, 0.0) end
end)

local ref = gui.Reference("Visuals", "Other")
local gb  = gui.Groupbox(ref, "Sun Info ESP", 325, 315, 300, 0)

local enable      = gui.Checkbox(gb, "sun.enable",   "Enable Sun ESP", true)
local show_on     = gui.Checkbox(gb, "sun.show_on",  "Show On/Off",    true)
local show_size   = gui.Checkbox(gb, "sun.show_size","Show Size",      true)
local show_color  = gui.Checkbox(gb, "sun.show_col", "Show Color",     true)

local font = draw.CreateFont("Tahoma", 13, 600)

local function UnpackColor(packed)
    local r = bit.band(packed, 0xFF)
    local g = bit.band(bit.rshift(packed, 8),  0xFF)
    local b = bit.band(bit.rshift(packed, 16), 0xFF)
    return r, g, b
end

callbacks.Register("Draw", "sun_info", function()
    if not enable:GetValue() then return end

    local suns = entities.FindByClass("CSun") or {}
    for _, sun in pairs(suns) do
        if sun:IsDormant() then goto continue end

        local lp = entities.GetLocalPlayer()
        if not lp then goto continue end

        local ok, dir = pcall(function() return sun:GetPropVector("m_vDirection") end)
        if not ok or not dir then goto continue end

        local eyePos = lp:GetAbsOrigin() + Vector3(0, 0, 64)
        local sunPos = eyePos + dir * 10000

        local sx, sy = client.WorldToScreen(sunPos)
        if not sx or not sy then goto continue end

        local lines = { "[ SUN ]" }

        if show_on:GetValue() then
            local on = sun:GetPropInt("m_bOn") or 0
            table.insert(lines, "On: " .. (on == 1 and "YES" or "NO"))
        end

        if show_size:GetValue() then
            local size = sun:GetPropInt("m_nSize") or 0
            table.insert(lines, "Size: " .. size)
        end

        if show_color:GetValue() then
            local clr = sun:GetPropInt("m_clrRender") or 0
            local r, g, b = UnpackColor(clr)
            table.insert(lines, "Color: " .. r .. ", " .. g .. ", " .. b)
        end

        draw.SetFont(font)
        for i, line in ipairs(lines) do
            local tw, _ = draw.GetTextSize(line)
            draw.Color(0, 0, 0, 180)
            draw.Text(sx - tw/2 + 1, sy + (i-1)*16 + 1, line)
            if i == 1 then
                draw.Color(255, 220, 50, 255)
            else
                draw.Color(255, 255, 255, 220)
            end
            draw.Text(sx - tw/2, sy + (i-1)*16, line)
        end

        ::continue::
    end
end)

local ref = gui.Reference("Misc", "General", "Extra")

local toneBox   = gui.Checkbox(ref, "custom_tonemap_active", "Tonemap", false)
local expSlider = gui.Slider(ref, "custom_tonemap_exposure", "Exposure", 3.0, 0.1, 10.0, 0.1)
local bloomSlider = gui.Slider(ref, "custom_tonemap_bloom", "Bloom", 1.0, 0.0, 10.0, 0.5)

callbacks.Register("Draw", "l4d2_tonemap_controller", function()
    if not toneBox:GetValue() then return end

    local tonemaps = entities.FindByClass("CEnvTonemapController")
    if not tonemaps then return end

    local exp = expSlider:GetValue()
    local bloom = bloomSlider:GetValue()

    for _, map in pairs(tonemaps) do
        map:SetPropInt(1, "m_bUseCustomAutoExposureMin")
        map:SetPropInt(1, "m_bUseCustomAutoExposureMax")
        map:SetPropInt(1, "m_bUseCustomBloomScale")

        map:SetPropFloat(exp, "m_flCustomAutoExposureMin")
        map:SetPropFloat(exp, "m_flCustomAutoExposureMax")
        map:SetPropFloat(bloom, "m_flCustomBloomScale")
    end
end)
