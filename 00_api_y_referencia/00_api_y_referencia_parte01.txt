local ref = gui.Reference("Misc", "General", "Extra")
local opt_block_plugins = gui.Checkbox(ref, "ffi.cl_showpluginmessages_mode", "Block Server Plugin Messages", false)

ffi.cdef[[
    void* GetModuleHandleA(const char* lpModuleName);
    void* GetProcAddress(void* hModule, const char* lpProcName);
]]

local FCVAR_CHEAT = 16384

local function GetInterface(dll, name)
    local module = ffi.C.GetModuleHandleA(dll)
    if module == nil then return nil end
    local ci_addr = ffi.C.GetProcAddress(module, "CreateInterface")
    if ci_addr == nil then return nil end
    local ci = ffi.cast("void*(__cdecl*)(const char*, int*)", ci_addr)
    return ci(name, nil)
end

local icvar = GetInterface("vstdlib.dll", "VEngineCvar007")
if not icvar then return end

local find_var_fn = ffi.cast("void*(__thiscall*)(void*, const char*)", ffi.cast("uintptr_t**", icvar)[0][13])

local function SetupConVar(name)
    local ptr = find_var_fn(icvar, name)
    if ptr == nil or ffi.cast("uintptr_t", ptr) == 0 then return nil end
    
    return {
        ptr = ptr,
        SetFloat    = ffi.cast("void(__thiscall*)(void*, float)", ffi.cast("uintptr_t**", ptr)[0][13]),
        RemoveFlags = ffi.cast("void(__thiscall*)(void*, int)", ffi.cast("uintptr_t**", ptr)[0][4])
    }
end

local cv_plugin_msg = SetupConVar("cl_showpluginmessages")

if cv_plugin_msg then 
    cv_plugin_msg.RemoveFlags(cv_plugin_msg.ptr, FCVAR_CHEAT) 
end

local last_state = -1

callbacks.Register("CreateMove", "engine_pluginmsg_logic", function(cmd)
    if not cv_plugin_msg then return end

    local state = opt_block_plugins:GetValue() and 0.0 or 1.0
    
    if state == last_state then return end
    last_state = state

    cv_plugin_msg.SetFloat(cv_plugin_msg.ptr, state)
end)

callbacks.Register("Unload", function()
    if cv_plugin_msg then 
        cv_plugin_msg.SetFloat(cv_plugin_msg.ptr, 1.0) 
    end
end)

===== END: Block Plugin Messages =====

local engineBase = ffi.cast("uintptr_t", ffi.C.GetModuleHandleA("engine.dll"))

local GetClientState = ffi.cast(
    "uintptr_t(__cdecl*)(void)",
    engineBase + 0x82CA0
)

local function GetMapName()
    local ok, cs = pcall(GetClientState)
    if not ok or cs == 0 then return nil end
    local ptr = ffi.cast("const char*", cs + 264)
    if ptr == nil then return nil end
    local ok2, name = pcall(ffi.string, ptr)
    if not ok2 then return nil end
    return name ~= "" and name or nil
end

ffi.cdef[[
    void* GetModuleHandleA(const char* lpModuleName);
    void* GetProcAddress(void* hModule, const char* lpProcName);
]]

local function GetInterface(dll, name)
    local mod = ffi.C.GetModuleHandleA(dll)
    if mod == nil then return nil end
    local ci = ffi.cast("void*(__cdecl*)(const char*, int*)",
        ffi.C.GetProcAddress(mod, "CreateInterface"))
    return ci(name, nil)
end

local baseClient = nil
for _, v in ipairs({"VClient017", "VClient016", "VClient015"}) do
    local ptr = GetInterface("client.dll", v)
    if ptr and ffi.cast("uintptr_t", ptr) ~= 0 then
        baseClient = ptr
        print("[ScreenHeight] Interface: " .. v)
        break
    end
end
if not baseClient then print("[ScreenHeight] baseClient alinamadi") return end

local vtable = ffi.cast("uintptr_t**", baseClient)[0]

local SLOT_GETSCREENHEIGHT = 60
local fnGetScreenHeight = ffi.cast("int(__thiscall*)(void*)", vtable[SLOT_GETSCREENHEIGHT])

local nLastPrint = 0

callbacks.Register("Draw", "screenheight_watch", function()
    local now = globals.CurTime()
    if now - nLastPrint < 1.0 then return end
    nLastPrint = now

    local ok, h = pcall(fnGetScreenHeight, baseClient)
    if ok then
        print("[ScreenHeight] " .. tostring(h))
    else
        print("[ScreenHeight] Hata: " .. tostring(h))
    end
end)

callbacks.Register("Unload", function()
    callbacks.Unregister("Draw", "screenheight_watch")
    print("[ScreenHeight] Kaldirildi")
end)

local ref = gui.Reference("Visuals", "World")
local group = gui.Groupbox(ref, "Hitbox Index Viewer", 320, 300, 305, 0)

local enable = gui.Checkbox(group, "hitbox.enable", "Enable Hitbox Viewer", false)
local colorpicker = gui.ColorPicker(group, "hitbox.color", "Hitbox Color", 255, 0, 0, 255)
local thickness = gui.Slider(group, "hitbox.thickness", "Line Thickness", 1, 1, 3)

local function ShowHitboxes()
    if not enable:GetValue() then return end

    local me = entities.GetLocalPlayer()
    if not me or not me:IsAlive() then return end

    local players = entities.FindByClass("CTerrorPlayer")
    if not players then return end

    local r, g, b, a = colorpicker:GetValue()
    draw.Color(r, g, b, a)

    for _, p in pairs(players) do
        if p == me then goto continue end
        if not p:IsAlive() or p:IsDormant() then goto continue end

        for i = 0, 20 do
            local x, y, z = p:GetHitboxPosition(i)
            if x then
                local sx, sy = client.WorldToScreen(x, y, z)
                if sx and sy then
                    draw.Text(sx, sy, tostring(i))
                    if thickness:GetValue() > 1 then
                        draw.Text(sx + 1, sy, tostring(i))
                    end
                end
            end
        end

        ::continue::
    end
end

callbacks.Register("Draw", ShowHitboxes)

local function UnloadHitboxes()
    callbacks.Unregister("Draw", ShowHitboxes)
    print("Unloaded cleanly.")
end

===== FILE: Index Finder for props =====
It can find all hittible indexes so u can detect it and just write a new lua for another prop esp. 



local done = false
callbacks.Register("Draw", "car_find", function()
    if done then return end
    done = true

    local lp = entities.GetLocalPlayer()
    if not lp then return end
    local mypos = lp:GetAbsOrigin()

    local classes = { "CPhysicsProp", "CBreakableProp", "CDynamicProp" }

    for _, cls in ipairs(classes) do
        local ents = entities.FindByClass(cls)
        if ents then
            for _, ent in pairs(ents) do
                if not ent:IsDormant() then
                    local pos = ent:GetAbsOrigin()
                    if pos then
                        local dx = pos.x - mypos.x
                        local dy = pos.y - mypos.y
                        local dz = pos.z - mypos.z
                        local dist = math.sqrt(dx*dx + dy*dy + dz*dz)
                        if dist < 150 then
                            local ok, idx = pcall(function()
                                return ent:GetPropInt("m_nModelIndex")
                            end)
                            print(string.format("[%s] dist=%.0f modelidx=%s", cls, dist, tostring(idx)))
                        end
                    end
                end
            end
        end
    end
    print("[DONE]")
end)

===== END: Index Finder for props =====

-- Map & Skybox Info

local font = draw.CreateFont("Tahoma", 14, 700)

local last_check = 0
local sky_name   = "unknown"
local map_name   = "unknown"

callbacks.Register("Draw", "sky_info", function()
    local now = globals.RealTime()

    if now - last_check > 2 then
        local ok1, v1 = pcall(function() return client.GetConVar("sv_skyname") end)
        local ok2, v2 = pcall(function() return client.GetConVar("host_map") end)

        if ok1 and v1 then sky_name = v1 end
        if ok2 and v2 then map_name = v2 end

        last_check = now
    end

    local ang = engine.GetViewAngles()
    if not ang or ang.x > -30 then return end

    local sw, sh = draw.GetScreenSize()

    draw.SetFont(font)

    local txt1 = "Map: " .. map_name
    local txt2 = "Sky: " .. sky_name

    local tw1, _ = draw.GetTextSize(txt1)
    local tw2, _ = draw.GetTextSize(txt2)

    local cx = sw / 2

    draw.Color(0, 0, 0, 180)
    draw.Text(cx - tw1/2 + 1, sh/2 - 30 + 1, txt1)
    draw.Text(cx - tw2/2 + 1, sh/2 - 10 + 1, txt2)

    draw.Color(100, 200, 255, 220)
    draw.Text(cx - tw1/2, sh/2 - 30, txt1)
    draw.Color(150, 220, 255, 220)
    draw.Text(cx - tw2/2, sh/2 - 10, txt2)
end)

callbacks.Register("Unload", function()
    callbacks.Unregister("Draw", "sky_info")
end)

===== END: Map and Skyname Info =====
