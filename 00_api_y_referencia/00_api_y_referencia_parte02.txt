local FCVAR_CHEAT = 16384

local function GetInterface(dll, name)
    local module = ffi.C.GetModuleHandleA(dll)
    if module == nil then return nil end
    local ci_addr = ffi.C.GetProcAddress(module, "CreateInterface")
    if ci_addr == nil then return nil end
    local ci = ffi.cast("void*(__cdecl*)(const char*, int*)", ci_addr)
    return ci(name, nil)
end

local icvar = GetInterface("vstdlib.dll", "VEngineCvar007")
if not icvar then return end

local find_var_fn = ffi.cast("void*(__thiscall*)(void*, const char*)", ffi.cast("uintptr_t**", icvar)[0][13])

local function SetupConVar(name)
    local ptr = find_var_fn(icvar, name)
    if ptr == nil or ffi.cast("uintptr_t", ptr) == 0 then return nil end
    
    return {
        ptr = ptr,
        -- ConVar VTable: SetFloat (13. indeks) ve RemoveFlags (4. indeks)
        SetFloat    = ffi.cast("void(__thiscall*)(void*, float)", ffi.cast("uintptr_t**", ptr)[0][13]),
        RemoveFlags = ffi.cast("void(__thiscall*)(void*, int)", ffi.cast("uintptr_t**", ptr)[0][4])
    }
end

local cv_overlay = SetupConVar("r_drawmodelstatsoverlay")

if cv_overlay then 
    cv_overlay.RemoveFlags(cv_overlay.ptr, FCVAR_CHEAT) 
end

local last_state = -1

callbacks.Register("CreateMove", "engine_overlay_logic", function(cmd)
    if not cv_overlay then return end

    local state = opt_overlay_esp:GetValue() and 1.0 or 0.0
    
    if state == last_state then return end
    last_state = state

    cv_overlay.SetFloat(cv_overlay.ptr, state)
end)

callbacks.Register("Unload", function()
    if cv_overlay then 
        cv_overlay.SetFloat(cv_overlay.ptr, 0.0) 
    end
end)

local function GetInterface(dll, name)
    local mod = ffi.C.GetModuleHandleA(dll)
    if mod == nil then return nil end
    local ci = ffi.cast("void*(__cdecl*)(const char*, int*)",

local MAX_CLASSES = 2000
local MAX_PROPS_PER_TABLE = 500
local MAX_RECURSE_DEPTH = 6


local function WalkTable(pTable, prefix, depth, outLines)
    if pTable == nil or depth > MAX_RECURSE_DEPTH then return end

    local nProps = pTable.m_nProps
    if nProps <= 0 or nProps > MAX_PROPS_PER_TABLE then return end

    for i = 0, nProps - 1 do
        local prop = pTable.m_pProps[i]
        if prop.m_pVarName ~= nil then
            local name = ffi.string(prop.m_pVarName)

            if prop.m_pDataTable ~= nil and name ~= "baseclass" then
           
                WalkTable(prop.m_pDataTable, prefix .. name .. ".", depth + 1, outLines)
            elseif name:sub(1, 2) == "m_" then
                table.insert(outLines, string.format("    %-40s offset=%-5d flags=%d",
                    prefix .. name, prop.m_Offset, prop.m_Flags))
            end
        end
    end
end


local pCur = pHead
local nCount = 0
local allLines = {}

while pCur ~= nil and nCount < MAX_CLASSES do
    local className = pCur.m_pNetworkName ~= nil and ffi.string(pCur.m_pNetworkName) or "?"
    local pTable = pCur.m_pRecvTable

    if pTable ~= nil then
        local lines = {}
        WalkTable(pTable, "", 0, lines)

        if #lines > 0 then
            table.insert(allLines, string.format("[%s]  (ClassID=%d)", className, pCur.m_ClassID))
            for _, l in ipairs(lines) do
                table.insert(allLines, l)
            end
        end
    end

    pCur = pCur.m_pNext
    nCount = nCount + 1
end

print(string.format("[NetVarDump] %d class tarandi, dokum asagida:", nCount))
for _, l in ipairs(allLines) do
    print(l)
end

function FindNetVarOffset(wantedClass, wantedProp)
    local pC = pHead
    local n = 0
    while pC ~= nil and n < MAX_CLASSES do
        local cname = pC.m_pNetworkName ~= nil and ffi.string(pC.m_pNetworkName) or ""
        if cname == wantedClass or cname == ("DT_" .. wantedClass) then
            local result = nil
            local function Search(pTable, depth)
                if pTable == nil or depth > MAX_RECURSE_DEPTH or result then return end
                for i = 0, pTable.m_nProps - 1 do
                    local prop = pTable.m_pProps[i]
                    if prop.m_pVarName ~= nil then
                        local pname = ffi.string(prop.m_pVarName)
                        if pname == wantedProp then
                            result = prop.m_Offset
                            return
                        elseif prop.m_pDataTable ~= nil then
                            Search(prop.m_pDataTable, depth + 1)
                        end
                    end
                end
            end
            Search(pC.m_pRecvTable, 0)
            if result then return result end
        end
        pC = pC.m_pNext
        n = n + 1
    end
    return nil
end

local MAX_RESULTS = 20

local function FindMatches(query)
    local results = {}

    if not query or query == "" then
        return results
    end

    query = query:lower()

    for _, api in ipairs(API_DB) do
        if api:lower():find(query, 1, true) then
            results[#results + 1] = api
            if #results >= MAX_RESULTS then
                break
            end
        end
    end

    return results
end

callbacks.Register("Draw", function()
    local q = search_box:GetValue()
    if q == "" then return end

    local matches = FindMatches(q)
    if #matches == 0 then return end

    local x = 30
    local y = 420
    local line_h = 15

    draw.Color(220, 220, 220, 255)

    for i, api in ipairs(matches) do
        draw.Text(x, y + (i - 1) * line_h, api)
    end
end)

    -- stateless browser, safe unload
end)
