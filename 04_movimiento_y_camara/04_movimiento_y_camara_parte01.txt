-- 180 Turn 
local ref = gui.Reference("Misc", "General", "Extra")
local gb  = gui.Groupbox(ref, "180 Turn", 295, 395, 300, 0)

local enable  = gui.Checkbox(gb, "t180.enable", "Enable 180 Turn", false)
local keybind = gui.Keybox(gb, "t180.key", "Turn Key", 0)

local was_pressed = false

callbacks.Register("CreateMove", "turn180", function(cmd)
    if not enable:GetValue() then return end

    local is_pressed = keybind:GetValue() ~= 0 and input.IsButtonDown(keybind:GetValue())

    if is_pressed and not was_pressed then
        local ang = engine.GetViewAngles()
        
        local new_yaw = ang.y + 180
        if new_yaw > 180 then new_yaw = new_yaw - 360 end
        if new_yaw < -180 then new_yaw = new_yaw + 360 end

        engine.SetViewAngles(EulerAngles(ang.x, new_yaw, 0))
    end

    was_pressed = is_pressed
end)

local function GetInterface(dll, name)
    local ok1, module = pcall(ffi.C.GetModuleHandleA, dll)
    if not ok1 or module == nil then return nil end
    
    local ok2, ci_addr = pcall(ffi.C.GetProcAddress, module, "CreateInterface")
    if not ok2 or ci_addr == nil then return nil end
    
    local ok3, ci = pcall(ffi.cast, "void*(__cdecl*)(const char*, int*)", ci_addr)
    if not ok3 or ci == nil then return nil end
    
    local ok4, result = pcall(ci, name, nil)
    if not ok4 then return nil end
    return result
end

local icvar = GetInterface("vstdlib.dll", "VEngineCvar007")

local ok_find, find_var_fn = pcall(function()
    if not icvar then return nil end
    return ffi.cast("void*(__thiscall*)(void*, const char*)", ffi.cast("uintptr_t**", icvar)[0][13])
end)

local function FindConVar(name)
    if not find_var_fn or not icvar then return nil end
    local ok, result = pcall(find_var_fn, icvar, name)
    if not ok then return nil end
    return result
end

local function RemoveCheatFlag(cv)
    if not cv then return end
    local ok, fn = pcall(function()
        return ffi.cast("void(__thiscall*)(void*, int)", ffi.cast("uintptr_t**", cv)[0][4])
    end)
    
    if ok and fn then
        pcall(fn, cv, 16384)
    end
end

local function SetFloat(cv, value)
    if not cv then return end
    local ok, fn = pcall(function()
        return ffi.cast("void(__thiscall*)(void*, float)", ffi.cast("uintptr_t**", cv)[0][13])
    end)
    
    if ok and fn then
        pcall(fn, cv, value)
    end
end

local cv_aspect = FindConVar("r_aspectratio")
if cv_aspect then

end

callbacks.Register("Draw", function()

end)

local font = draw.CreateFont("Tahoma", 15, 800)
local hits  = {}

client.AllowListener("player_falldamage")

callbacks.Register("FireGameEvent", function(event)
    if event:GetName() ~= "player_falldamage" then return end
    local userid = event:GetInt("userid")
    local dmg    = event:GetInt("damage")
    local ent    = entities.GetByIndex(client.GetPlayerIndexByUserID(userid))
    if not ent then return end
    local pos = ent:GetAbsOrigin()
    if not pos then return end
    table.insert(hits, {
        pos    = pos,
        dmg    = dmg,
        time   = globals.RealTime(),
        expire = globals.RealTime() + 3.5,
    })
end)

callbacks.Register("Draw", "falldmg_draw", function()
    local now = globals.RealTime()
    local i   = 1
    while i <= #hits do
        local h = hits[i]
        if now > h.expire then
            table.remove(hits, i)
        else
            local age    = now - h.time
            local fade_start = 2.0
            local alpha
            if age < fade_start then
                alpha = 255
            else
                alpha = math.floor(255 * (1 - (age - fade_start) / (3.5 - fade_start)))
                alpha = math.max(0, alpha)
            end
            local floatY = math.floor(age * 25)

            local sx, sy = client.WorldToScreen(h.pos + Vector3(0,0,60))
            if sx and sy then
                local txt = "-" .. h.dmg
                draw.SetFont(font)
                local tw, _ = draw.GetTextSize(txt)
                draw.Color(0,0,0,math.floor(alpha*0.6))
                draw.Text(sx-tw/2+1, sy-floatY+1, txt)
                draw.Color(255, 140, 0, alpha)
                draw.Text(sx-tw/2, sy-floatY, txt)
            end
            i = i + 1
        end
    end
end)

===== END: Fall Damage Viewer =====

local enabledddcheckbosx = gui.Checkbox(gui.Reference("Misc", "General", "Extra"), "fastladder", "Fast Ladders", 0)

local function PORNAAAA(cmd)
    if not gui.GetValue("misc.master") then return end
    if not enabledddcheckbosx:GetValue() then return end
    
    local LP = entities.GetLocalPlayer()
    if not LP then return end

    if bit.band(LP:GetPropInt("m_nRenderMode"), bit.lshift(1, 8)) == 0 then return end 
    
    local laddbasic = LP:GetPropVector("m_vecLadderNormal")

    if cmd.forwardmove ~= 0 then
        if cmd.forwardmove > 0 then
            cmd.buttons = bit.band(bit.bor(cmd.buttons, bit.lshift(1, 9)), bit.bnot(bit.lshift(1, 10))) 
        else
            cmd.buttons = bit.band(bit.bor(cmd.buttons, bit.lshift(1, 10)), bit.bnot(bit.lshift(1, 9))) 
        end
        
        local angleshit = EulerAngles(-89, math.deg(math.acos(laddbasic.x)) + (laddbasic.y < 0 and 180 or 0) + 90, 0)
        angleshit:Normalize()
        cmd.viewangles = angleshit
    end
end

callbacks.Register("CreateMove", PORNAAAA)

-- freecam
local ref             = gui.Reference("Misc", "General", "Extra")
local freecam_enable  = gui.Checkbox(ref, "misc.freecam",       "Free Camera", false)
local freecam_speed   = gui.Slider(ref,  "misc.freecam_speed",  "Cam Speed",   5, 1, 20, 1)


ffi.cdef[[
    void* GetModuleHandleA(const char* lpModuleName);
    void* GetProcAddress(void* hModule, const char* lpProcName);
    int   VirtualProtect(void* lpAddress, size_t dwSize, uint32_t flNewProtect, uint32_t* lpflOldProtect);
]]

local PAGE_EXECUTE_READWRITE = 0x40
local OVERRIDE_VIEW_INDEX    = 19

local VS_ORIGIN_X, VS_ORIGIN_Y, VS_ORIGIN_Z = 0x3C, 0x40, 0x44
local VS_ANGLES_X, VS_ANGLES_Y, VS_ANGLES_Z = 0x48, 0x4C, 0x50

local cm_sig = mem.FindPattern("client.dll", "89 04 B5 ?? ?? ?? ?? E8")
if not cm_sig then print("freecam CM sig fail") return end

local global_addr = ffi.cast("uintptr_t*", ffi.cast("uintptr_t", cm_sig) + 3)[0]
if global_addr == 0 then print("[FreeCam] Global null") return end

local clientmode = ffi.cast("uintptr_t*", global_addr)[0]
if clientmode == 0 then print("freecam CM null") return end

print(string.format("freecam CM: 0x%X", tonumber(clientmode)))

local vtable   = ffi.cast("uintptr_t*", ffi.cast("uintptr_t*", clientmode)[0])
local slot_ptr = vtable + OVERRIDE_VIEW_INDEX

local GC = { original = slot_ptr[0], hook_fn = nil, slot_ptr = slot_ptr }
print(string.format("freecam Orig: 0x%X", tonumber(GC.original)))


local function ReadFloat(ptr, offset)
    return ffi.cast("float*", ptr + offset)[0]
end

local function WriteFloat(ptr, offset, val)
    ffi.cast("float*", ptr + offset)[0] = val
end

local function AngleVectors(ax, ay)
    local p, y = math.rad(ax), math.rad(ay)
    local sp, cp, sy, cy = math.sin(p), math.cos(p), math.sin(y), math.cos(y)
    return { x = cp*cy, y = cp*sy, z = -sp }, { x = sy, y = -cy, z = 0 }
end

local cam_pos    = nil
local IN_FORWARD, IN_BACK, IN_MOVELEFT, IN_MOVERIGHT = 8, 16, 512, 1024
local IN_JUMP, IN_DUCK = 2, 4

local ov_t = ffi.typeof("void(__fastcall*)(void*, void*, void*)")

local function Hooked_OverrideView(ecx, edx, pSetup)
    if GC.original == nil then return end
    
    ffi.cast(ov_t, GC.original)(ecx, edx, pSetup)

    if not freecam_enable:GetValue() then
        cam_pos = nil
        return
    end

    local ptr = ffi.cast("uintptr_t", pSetup)
    if ptr == 0 then return end

    if cam_pos == nil then
        cam_pos = {
            x = ReadFloat(ptr, VS_ORIGIN_X),
            y = ReadFloat(ptr, VS_ORIGIN_Y),
            z = ReadFloat(ptr, VS_ORIGIN_Z) + 80,
        }
        print(string.format("freecam Start: %.1f %.1f %.1f", cam_pos.x, cam_pos.y, cam_pos.z))
    end

    WriteFloat(ptr, VS_ORIGIN_X, cam_pos.x)
    WriteFloat(ptr, VS_ORIGIN_Y, cam_pos.y)
    WriteFloat(ptr, VS_ORIGIN_Z, cam_pos.z)
end

GC.hook_fn = ffi.cast(ov_t, Hooked_OverrideView)

local old = ffi.new("uint32_t[1]")
ffi.C.VirtualProtect(ffi.cast("void*", slot_ptr), 4, PAGE_EXECUTE_READWRITE, old)
slot_ptr[0] = ffi.cast("uintptr_t", GC.hook_fn)
ffi.C.VirtualProtect(ffi.cast("void*", slot_ptr), 4, old[0], old)
print("freecam hooked")

callbacks.Register("CreateMove", "freecam_move", function(cmd)
    if not freecam_enable:GetValue() or cam_pos == nil then return end

    local ang = engine.GetViewAngles()
    local fwd, right = AngleVectors(ang.x, ang.y)
    local speed = freecam_speed:GetValue()

    if bit.band(cmd.buttons, IN_FORWARD)   ~= 0 then
        cam_pos.x = cam_pos.x + fwd.x * speed
        cam_pos.y = cam_pos.y + fwd.y * speed
        cam_pos.z = cam_pos.z + fwd.z * speed
    end
    if bit.band(cmd.buttons, IN_BACK)      ~= 0 then
        cam_pos.x = cam_pos.x - fwd.x * speed
        cam_pos.y = cam_pos.y - fwd.y * speed
        cam_pos.z = cam_pos.z - fwd.z * speed
    end
    if bit.band(cmd.buttons, IN_MOVERIGHT) ~= 0 then
        cam_pos.x = cam_pos.x + right.x * speed
        cam_pos.y = cam_pos.y + right.y * speed
    end
    if bit.band(cmd.buttons, IN_MOVELEFT)  ~= 0 then
        cam_pos.x = cam_pos.x - right.x * speed
        cam_pos.y = cam_pos.y - right.y * speed
    end
    if bit.band(cmd.buttons, IN_JUMP)      ~= 0 then cam_pos.z = cam_pos.z + speed end
    if bit.band(cmd.buttons, IN_DUCK)      ~= 0 then cam_pos.z = cam_pos.z - speed end

    cmd.viewangles, cmd.forwardmove, cmd.sidemove, cmd.upmove, cmd.buttons = ang, 0, 0, 0, 0
end)

callbacks.Register("Unload", function()
    if GC.original and GC.slot_ptr then
        local old2 = ffi.new("uint32_t[1]")
        ffi.C.VirtualProtect(ffi.cast("void*", GC.slot_ptr), 4, PAGE_EXECUTE_READWRITE, old2)
        GC.slot_ptr[0] = ffi.cast("uintptr_t", GC.original)
        ffi.C.VirtualProtect(ffi.cast("void*", GC.slot_ptr), 4, old2[0], old2)
        print("freecam unhooked")
    end
    cam_pos = nil 
end)

===== END: Freecam =====
