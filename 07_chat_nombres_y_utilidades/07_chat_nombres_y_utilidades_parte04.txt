ffi.cdef[[
    void* GetModuleHandleA(const char* lpModuleName);
    int   VirtualProtect(void* lpAddress, size_t dwSize, uint32_t flNewProtect, uint32_t* lpflOldProtect);
]]
local PAGE_EXECUTE_READWRITE = 0x40
local engine_base = ffi.cast("uintptr_t", ffi.C.GetModuleHandleA("engine.dll"))
if engine_base == 0 then
    print("[VANILLA] engine.dll bulunamadi")
    return
end
print(string.format("[VANILLA] engine.dll base: 0x%X", tonumber(engine_base)))
local PATCH_OFFSET = 0x5596C  
local patch_addr   = ffi.cast("uint8_t*", engine_base + PATCH_OFFSET)
local b0, b1, b2, b3 = patch_addr[0], patch_addr[1], patch_addr[2], patch_addr[3]
print(string.format("[VANILLA] Adresteki byte'lar: %02X %02X %02X %02X", b0, b1, b2, b3))
if b0 ~= 0x0F or b1 ~= 0xB6 or b2 ~= 0x4F or b3 ~= 0x5D then
    print("[VANILLA] HATA: Beklenen 0F B6 4F 5D degil!")
    print("[VANILLA] Offset yanlis olabilir — kontrol et")
    return
end
print("[VANILLA] Byte dogrulama OK (0F B6 4F 5D)")
local orig_bytes = ffi.new("uint8_t[4]")
for i = 0, 3 do orig_bytes[i] = patch_addr[i] end
local old = ffi.new("uint32_t[1]")
ffi.C.VirtualProtect(patch_addr, 4, PAGE_EXECUTE_READWRITE, old)
patch_addr[0] = 0x31  
patch_addr[1] = 0xC9  
patch_addr[2] = 0x90 
patch_addr[3] = 0x90  
ffi.C.VirtualProtect(patch_addr, 4, old[0], old)
print("[VANILLA] Patch uygulandi!")
print("[VANILLA] Sunucu vanilla=1 gondersenin artik onemi yok — client hep vanilla=0 gorecek")
_G.vanilla_GC = { addr = patch_addr, orig = orig_bytes }
function on_unload()
    local g = _G.vanilla_GC
    if not g then return end
    local o2 = ffi.new("uint32_t[1]")
    ffi.C.VirtualProtect(g.addr, 4, PAGE_EXECUTE_READWRITE, o2)
    for i = 0, 3 do g.addr[i] = g.orig[i] end
    ffi.C.VirtualProtect(g.addr, 4, o2[0], o2)
    _G.vanilla_GC = nil
    print("[VANILLA] Unloaded, orijinal byte'lar restore edildi.")
end

===== FILE: Fake Chat Message =====
pcall(function()
    ffi.cdef[[
        void* GetModuleHandleA(const char* lpModuleName);
    ]]
end)

local cm_sig = mem.FindPattern("client.dll", "89 04 B5 ?? ?? ?? ?? E8")
if not cm_sig then print("sig bulunamadi") return end

local function GetCHudChat()
    local ok, result = pcall(function()
        local global_addr = ffi.cast("uintptr_t*", ffi.cast("uintptr_t", cm_sig) + 3)[0]
        if global_addr == 0 then return nil end
        local clientmode = ffi.cast("uintptr_t*", global_addr)[0]
        if clientmode == 0 then return nil end
        local chudchat = ffi.cast("uintptr_t*", clientmode + 0x18)[0]
        if chudchat == 0 then return nil end
        return chudchat
    end)
    if not ok or result == nil then return nil end
    return result
end

local function ChatPrintf(idx, msg)
    local chudchat = GetCHudChat()
    if not chudchat then return end
    pcall(function()
        local vtable = ffi.cast("uintptr_t**", chudchat)[0]
        local fn     = ffi.cast("void(__stdcall*)(void*, int, int, const char*)", vtable[23])
        fn(ffi.cast("void*", chudchat), idx, 0, msg)
    end)
end

local ref     = gui.Reference("Misc", "General", "Extra")
local nameBox = gui.Editbox(ref, "fakechat.name", "Nick")
local msgBox  = gui.Editbox(ref, "fakechat.msg",  "Mesaj")
local sendKey = gui.Keybox(ref,  "fakechat.key",  "Gönder", 0)
local teamBox = gui.Checkbox(ref, "fakechat.team", "Team Chat", false)

local wasDown = false

callbacks.Register("Draw", "fakechat", function()
    local key = sendKey:GetValue()
    if key == 0 then return end

    local down = input.IsButtonDown(key)
    if not down then wasDown = false return end
    if wasDown then return end
    wasDown = true

    local name   = nameBox:GetValue()
    local msg    = msgBox:GetValue()
    local prefix = teamBox:GetValue() and "\x05" or "\x03"

    ChatPrintf(0, string.format("%s%s\x01: %s", prefix, name, msg))
end)

callbacks.Register("Unload", function()
    callbacks.Unregister("Draw", "fakechat")
end)

===== FILE: Fake VC =====
Voice Status hook



pcall(function()
    ffi.cdef[[
        void* GetModuleHandleA(const char* lpModuleName);
        void* GetProcAddress(void* module, const char* name);
    ]]
end)

local clientDll = ffi.C.GetModuleHandleA("client.dll")
local createInterface = ffi.cast(
    "void*(__cdecl*)(const char*, int*)",
    ffi.C.GetProcAddress(clientDll, "CreateInterface")
)

local baseClient = nil
for _, v in ipairs({"VClient017", "VClient016", "VClient015"}) do
    local ptr = createInterface(v, nil)
    if ptr and ffi.cast("uintptr_t", ptr) ~= 0 then
        baseClient = ptr
        print("[VoiceStatus] Interface:", v)
        break
    end
end

if not baseClient then
    print("[VoiceStatus] IBaseClientDLL alinamadi")
    return
end

local vtable    = ffi.cast("uintptr_t**", baseClient)[0]
local voiceFn   = ffi.cast("void(__thiscall*)(void*, int, int, int)", vtable[32])

local ref       = gui.Reference("Misc", "General", "Extra")
local allBox    = gui.Checkbox(ref, "voice.all",  "Voice - Herkes",       false)
local teamBox   = gui.Checkbox(ref, "voice.team", "Voice - Takımım",      false)
local enemyBox  = gui.Checkbox(ref, "voice.enemy","Voice - Karşı Takım",  false)

local function ApplyVoice()
    local pLocal = entities.GetLocalPlayer()
    if not pLocal then return end
    local localTeam = pLocal:GetTeamNumber()

    for i = 1, globals.MaxClients() do
        local ent = entities.GetByIndex(i)
        if ent and not ent:IsDormant() then
            local talking = false

            if allBox:GetValue() then
                talking = true
            elseif teamBox:GetValue() and ent:GetTeamNumber() == localTeam then
                talking = true
            elseif enemyBox:GetValue() and ent:GetTeamNumber() ~= localTeam then
                talking = true
            end

            pcall(voiceFn, baseClient, i, 0, talking and 1 or 0)
        end
    end
end

callbacks.Register("Draw", "voicestatus", ApplyVoice)

callbacks.Register("Unload", function()
    for i = 1, globals.MaxClients() do
        pcall(voiceFn, baseClient, i, 0, 0)
    end
    callbacks.Unregister("Draw", "voicestatus")
end)
