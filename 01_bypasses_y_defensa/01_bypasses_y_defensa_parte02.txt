local function write_bytes(addr, bytes)
    local ptr = ffi.cast("uint8_t*", addr)
    local old = ffi.new("uint32_t[1]")
    ffi.C.VirtualProtect(ffi.cast("void*", addr), #bytes, 0x40, old)
    for i, b in ipairs(bytes) do
        ptr[i-1] = b
    end
    ffi.C.VirtualProtect(ffi.cast("void*", addr), #bytes, old[0], old)
end

local FCVAR_CHEAT = 16384

local function GetInterface(dll, name)
    local module = ffi.C.GetModuleHandleA(dll)
    local ci_addr = ffi.C.GetProcAddress(module, "CreateInterface")
    local ci = ffi.cast("void*(__cdecl*)(const char*, int*)", ci_addr)
    return ci(name, nil)
end

local icvar = GetInterface("vstdlib.dll", "VEngineCvar007")
local find_var_fn = ffi.cast("void*(__thiscall*)(void*, const char*)", ffi.cast("uintptr_t**", icvar)[0][13])

local function SetupConVar(name)
    local ptr = find_var_fn(icvar, name)
    if ptr == nil or ffi.cast("uintptr_t", ptr) == 0 then return nil end
    
    return {
        ptr = ptr,
        SetInt      = ffi.cast("void(__thiscall*)(void*, int)", ffi.cast("uintptr_t**", ptr)[0][16]),
        RemoveFlags = ffi.cast("void(__thiscall*)(void*, int)", ffi.cast("uintptr_t**", ptr)[0][4])
    }
end

local function GetSteamPath()
    if not Advapi32 then
        cFullSteamPath = cBackupSteamPath
        return
    end

    local hKey = ffi.new("void*[1]")
    local ok, res = pcall(function()
        return Advapi32.RegOpenKeyExA(HKEY_CURRENT_USER, HKEY_STEAM_SUB_PATH, 0, KEY_QUERY_VALUE, hKey)
    end)

    if ok and res == ERROR_SUCCESS then
        local buf  = ffi.new("unsigned char[1024]")
        local bsz  = ffi.new("unsigned long[1]", 1024)
        ok, res = pcall(function()
            return Advapi32.RegQueryValueExA(hKey[0], "SteamExe", nil, nil, buf, bsz)
        end)
        cFullSteamPath = (ok and res == ERROR_SUCCESS)
            and ffi.string(buf):gsub("/", "\\")
            or cBackupSteamPath
        pcall(function() Advapi32.RegCloseKey(hKey[0]) end)
    else
        cFullSteamPath = cBackupSteamPath
    end
end

local function CreateActionFile(fileName)
    if not Kernel32 then return false end
    local fullPath = TempBridgePath .. '\\' .. fileName
    local ok, hFile = pcall(function()
        return Kernel32.CreateFileA(fullPath, GENERIC_ALL, 0, nil, CREATE_ALWAYS, FILE_ATTRIBUTE_NORMAL, nil)
    end)
    if ok and hFile ~= INVALID_HANDLE_VALUE then
        pcall(function() Kernel32.CloseHandle(hFile) end)
        return true
    end
    print("[RCB] Dosya olusturulamadi: " .. fullPath)
    return false
end


local function EnsureInit()
    if not bPowerShellWasInit then
        GetSteamPath()
        if InitPowerShellScript() then
            bPowerShellWasInit = true
        end
    end
    return bPowerShellWasInit
end
